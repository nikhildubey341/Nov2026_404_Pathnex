Day 12 - Scaling Applications and Automation
# Ansible - Setup Auto-Scaling Group (ASG)
- name: Setup Auto Scaling Group                     # name (play) = a label for the whole job
  hosts: all                                         # hosts: all = run on every server in the inventory file
  become: yes                                        # become: yes = run as admin (sudo)
  tasks:                                             # tasks = the list of steps to run in order
    - name: Create Launch Configuration              # name (task) = a label for one step
      ec2_asg:                                       # ec2_asg = module that manages an AWS Auto Scaling Group
        name: pathnex-asg                            # name (ec2_asg module) = the name of the Auto Scaling Group
        region: us-east-1                            # region = the data center location to use
        launch_config_name: pathnex-launch-config    # launch_config_name = the template used to create servers
        min_size: 1                                  # min_size = the fewest servers allowed
        max_size: 3                                  # max_size = the most servers allowed
        desired_capacity: 2                          # desired_capacity = how many servers to run normally
        vpc_zone_identifier: subnet-12345678         # vpc_zone_identifier = the subnets where servers are placed
		
# Terraform - Create a Load Balancer with EC2 Instances
resource "aws_lb" "pathnex_lb" {                     # resource = a thing Terraform creates (type + my own nickname)
  name               = "pathnex-lb"                  # name (load balancer) = the name of the load balancer
  internal           = false                         # internal = true keeps it private, false makes it public
  load_balancer_type = "application"                 # load_balancer_type = the kind of load balancer (application = works with web traffic)
  security_groups   = [aws_security_group.allow_ssh_http.id]   # security_groups = the security groups attached to the server
  subnets           = [aws_subnet.public.id]         # subnets = the subnets the load balancer is placed in
}

resource "aws_instance" "pathnex_ec2" {
  ami           = "ami-0abcd1234abcd1234"            # ami = the OS image used to create the server
  instance_type = "t2.micro"                         # instance_type = the size of the server (CPU and RAM)
  security_groups = [aws_security_group.allow_ssh_http.name]
  tags = {                                           # tags = labels used to find and organise the server
    Name = "Pathnex-EC2"                             # Name = the name shown in the AWS console
  }
}

# Kubernetes - Horizontal Pod Autoscaler (HPA) with Load Balancer
apiVersion: apps/v1                                  # apiVersion = which Kubernetes API version this object uses
kind: Deployment                                     # kind = the type of object to create
metadata:                                            # metadata = information that identifies this object
  name: pathnex-web-app                              # name (Kubernetes) = the name of this object
spec:                                                # spec = the desired setup of this object
  replicas: 2                                        # replicas = how many pods should always be running
  selector:                                          # selector = how it finds the pods that belong to it
    matchLabels:                                     # matchLabels = pods with this label belong to it
      app: pathnex-web-app
  template:                                          # template = the blueprint used to create each pod
    metadata:
      labels:                                        # labels = tags attached to the pod (must match the selector)
        app: pathnex-web-app
    spec:
      containers:                                    # containers = the list of containers inside the pod
        - name: web                                  # name (container) = the name of the container inside the pod
          image: nginx                               # image = the Docker image to run
          ports:                                     # ports = the list of ports the container uses
            - containerPort: 80                      # containerPort = the port the app listens on inside the container
			
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: pathnex-web-app-hpa
spec:
  scaleTargetRef:                                    # scaleTargetRef = the object to scale
    apiVersion: apps/v1
    kind: Deployment
    name: pathnex-web-app                            # name (scale target) = the name of the object to scale
  minReplicas: 1                                     # minReplicas = the fewest pods allowed
  maxReplicas: 5                                     # maxReplicas = the most pods allowed
  targetCPUUtilizationPercentage: 80                 # targetCPUUtilizationPercentage = the CPU usage % that triggers scaling
  
# Jenkinsfile - Implement Parallel Stages
pipeline {                                           # pipeline = the whole CI/CD process
    agent any                                        # agent any = run on any available machine
    stages {                                         # stages (Jenkins) = the list of phases
        stage('Build & Test') {                      # stage (Jenkins) = one phase
            parallel {                               # parallel = runs the stages inside it at the same time
                stage('Build') {
                    steps {                          # steps = the commands to run in a stage
                        echo 'Building project...'   # echo = prints the given text on the screen
                    }
                }
                stage('Test') {
                    steps {
                        echo 'Running tests...'
                    }
                }
            }
        }
        stage('Deploy') {
            steps {
                echo 'Deploying project...'
            }
        }
    }
}

# GitLab CI/CD - Parallel Jobs for Build & Test
stages:                                              # stages (GitLab) = the order in which jobs run
  - build                                            # build = the stage that builds
  - test                                             # test = the stage that tests
  - deploy                                           # deploy = the stage that releases

build:                                               # job name = a job GitLab runs
  stage: build                                       # stage: = the stage this job belongs to
  script:                                            # script (GitLab) = the commands the job runs
    - docker build -t pathnex-web-app .              # docker build = builds an image from the Dockerfile | -t = gives the image a name (tag) | . = the current folder

test:
  stage: test
  script:
    - docker run pathnex-web-app npm test            # docker run = create and start a container from an image | npm test = runs the project's tests

deploy:
  stage: deploy
  script:
    - kubectl apply -f kubernetes/deployment.yaml    # kubectl apply -f = creates or updates objects from a file
